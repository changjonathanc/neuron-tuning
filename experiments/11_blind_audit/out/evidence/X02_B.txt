import socket

def simple_client(host, port, message_to_send):
    """
    Establishes a TCP connection to a server, sends a message, receives a response,
    prints the response, and closes the connection.

    Args:
        host (str): The hostname or IP address of the server.
        port (int): The port number of the server.
        message_to_send (str): The message string to send to the server.
    """
    # 1. Create a socket object
    try:
        client_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        print(f"Socket created successfully.")

        # 2. Establish the connection
        print(f"Attempting to connect to {host}:{port}...")
        client_socket.connect((host, port))
        print("Connection established successfully.")

        # 3. Send the message
        print(f"Sending message: '{message_to_send}'")
        # Messages must be encoded to bytes before sending
        client_socket.sendall(message_to_send.encode('utf-8'))

        # 4. Receive the response
        # We set a buffer size (e.g., 1024 bytes) to receive data
        data = client_socket.recv(1024)
        
        # Decode the received bytes back into a string
        response = data.decode('utf-8')

        # 5. Print the response
        print("\n--- Server Response ---")
        print(response)
        print("-----------------------\n")

    except ConnectionRefusedError:
        print(f"Error: Connection refused. Ensure a server is running at {host}:{port}.")
    except socket.gaierror:
        print(f"Error: Address resolution failed. Check the hostname '{host}'.")
    except socket.error as e:
        print(f"An unexpected socket error occurred: {e}")
    finally:
        # 6. Close the connection
        if 'client_socket' in locals() and client_socket:
            print("Closing the connection.")
            client_socket.close()
